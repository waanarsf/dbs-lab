1. Based on the University database Schema in Lab 2, write a row trigger that records 
along with the time any change made in the Takes (ID, course-id, sec-id, semester, 
year, grade) table in log_change_Takes (Time_Of_Change, ID, courseid,sec-id, 
semester, year, grade). 
->
CREATE TABLE log_change_Takes (
    Time_Of_Change TIMESTAMP,
    ID             VARCHAR(5),
    course_id      VARCHAR(8),
    sec_id         VARCHAR(8),
    semester       VARCHAR(6),
    year           NUMERIC(4,0),
    grade          VARCHAR(2)
);
CREATE OR REPLACE TRIGGER trg_log_takes_change
AFTER INSERT OR UPDATE OR DELETE ON takes
FOR EACH ROW
BEGIN
    IF DELETING THEN
        INSERT INTO log_change_Takes (Time_Of_Change, ID, course_id, sec_id, semester, year, grade)
        VALUES (SYSTIMESTAMP, :OLD.ID, :OLD.course_id, :OLD.sec_id, :OLD.semester, :OLD.year, :OLD.grade);
    ELSE
        -- For INSERT or UPDATE, we log the new state of the row
        INSERT INTO log_change_Takes (Time_Of_Change, ID, course_id, sec_id, semester, year, grade)
        VALUES (SYSTIMESTAMP, :NEW.ID, :NEW.course_id, :NEW.sec_id, :NEW.semester, :NEW.year, :NEW.grade);
    END IF;
END;
/
UPDATE takes 
SET grade = 'A+' 
WHERE ID = '00128' AND course_id = 'CS-101';
SELECT * FROM log_change_Takes;


2. Based on the University database schema in Lab: 2, write a row trigger to insert 
the existing values of the Instructor (ID, name, dept-name, salary) table into a new 
table Old_ Data_Instructor (ID, name, dept-name, salary) when the salary table is 
updated.

->
CREATE TABLE Old_Data_Instructor (
    ID          VARCHAR(5),
    name        VARCHAR(20),
    dept_name   VARCHAR(20),
    salary      NUMERIC(8,2)
);
CREATE OR REPLACE TRIGGER trg_archive_old_salary
BEFORE UPDATE OF salary ON instructor
FOR EACH ROW
BEGIN
    -- :OLD refers to the data currently in the table before the update
    INSERT INTO Old_Data_Instructor (ID, name, dept_name, salary)
    VALUES (:OLD.ID, :OLD.name, :OLD.dept_name, :OLD.salary);
END;
/
-- Giving Einstein a raise from 95,000 to 100,000
UPDATE instructor 
SET salary = 100000 
WHERE name = 'Einstein';
SELECT * FROM Old_Data_Instructor;

3. Based on the University Schema, write a database trigger on Instructor that checks 
the following: 
 The name of the instructor is a valid name containing only alphabets. 
 The salary of an instructor is not zero and is positive 
 The salary does not exceed the budget of the department to which the 
instructor belongs.
->
CREATE OR REPLACE TRIGGER trg_validate_instructor
BEFORE INSERT OR UPDATE ON instructor
FOR EACH ROW
DECLARE
    v_dept_budget NUMBER(12,2);
BEGIN
    -- 1. Validate Name: Contains only alphabets (and spaces)
    -- REGEXP_LIKE returns true if the name matches the pattern
    IF NOT REGEXP_LIKE(:NEW.name, '^[A-Za-z ]+$') THEN
        RAISE_APPLICATION_ERROR(-20001, 'Invalid Name: Instructor name must contain only alphabets.');
    END IF;

    -- 2. Validate Salary: Must be positive and non-zero
    IF :NEW.salary <= 0 THEN
        RAISE_APPLICATION_ERROR(-20002, 'Invalid Salary: Salary must be a positive, non-zero value.');
    END IF;

    -- 3. Validate Salary vs Department Budget
    -- We need to fetch the budget of the specific department assigned to the instructor
    SELECT budget INTO v_dept_budget 
    FROM department 
    WHERE dept_name = :NEW.dept_name;

    IF :NEW.salary > v_dept_budget THEN
        RAISE_APPLICATION_ERROR(-20003, 'Budget Exceeded: Instructor salary (' || :NEW.salary || 
                                ') cannot exceed department budget (' || v_dept_budget || ').');
    END IF;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        -- Handles cases where the dept_name might not exist in the department table
        RAISE_APPLICATION_ERROR(-20004, 'Department Not Found: The assigned department does not exist.');
END;
/
-- This should FAIL
INSERT INTO instructor VALUES ('99999', 'Einstein123', 'Physics', 95000);
-- This should FAIL
UPDATE instructor SET salary = 0 WHERE ID = '10101';
-- This should FAIL because 95,000 > 90,000
UPDATE instructor SET salary = 95000 WHERE ID = '76766';


4. Create a transparent audit system for a table Client_master (client_no, name, 
address, Bal_due). The system must keep track of the records that are being 
deleted or updated. The functionality being when a record is deleted or modified 
the original record details and the date of operation are stored in the auditclient 
(client_no, name, bal_due, operation, userid, opdate) table, then the delete or 
update is allowed to go through. 
->

-- The main table
CREATE TABLE Client_master (
    client_no   VARCHAR2(6) PRIMARY KEY,
    name        VARCHAR2(20),
    address     VARCHAR2(30),
    Bal_due     NUMBER(10,2)
);

-- The audit table
CREATE TABLE auditclient (
    client_no   VARCHAR2(6),
    name        VARCHAR2(20),
    bal_due     NUMBER(10,2),
    operation   VARCHAR2(10), -- Stores 'UPDATE' or 'DELETE'
    userid      VARCHAR2(30), -- Stores the database user who made the change
    opdate      DATE          -- Stores the date/time of the operation
);
CREATE OR REPLACE TRIGGER trg_audit_client
BEFORE UPDATE OR DELETE ON Client_master
FOR EACH ROW
DECLARE
    v_operation VARCHAR2(10);
BEGIN
    -- Determine which operation is being performed
    IF UPDATING THEN
        v_operation := 'UPDATE';
    ELSIF DELETING THEN
        v_operation := 'DELETE';
    END IF;

    -- Insert the ORIGINAL (old) values into the audit table
    INSERT INTO auditclient (
        client_no, 
        name, 
        bal_due, 
        operation, 
        userid, 
        opdate
    ) VALUES (
        :OLD.client_no, 
        :OLD.name, 
        :OLD.Bal_due, 
        v_operation, 
        USER, 
        SYSDATE
    );
END;
/
-- Assuming a client exists with client_no 'C001'
DELETE FROM Client_master WHERE client_no = 'C001';
 
5. Based on the University database Schema in Lab 2, create a view Advisor_Student 
which is a natural join on Advisor, Student and Instructor tables. Create an 
INSTEAD OF trigger on Advisor_Student to enable the user to delete the 
corresponding entries in Advisor table. 
->
CREATE OR REPLACE VIEW Advisor_Student AS
SELECT 
    s.ID AS student_id, 
    s.name AS student_name, 
    i.ID AS instructor_id, 
    i.name AS instructor_name,
    s.dept_name
FROM student s
JOIN advisor a ON s.ID = a.s_ID
JOIN instructor i ON i.ID = a.i_ID;

CREATE OR REPLACE TRIGGER trg_delete_advisor_student
INSTEAD OF DELETE ON Advisor_Student
FOR EACH ROW
BEGIN
    -- We delete only from the bridge table (Advisor)
    -- using the student ID from the view row being deleted.
    DELETE FROM advisor 
    WHERE s_ID = :OLD.student_id;
END;
/
SELECT * FROM Advisor_Student WHERE student_name = 'Zhang';
-- This would normally throw an error without the trigger
DELETE FROM Advisor_Student 
WHERE student_name = 'Zhang';
